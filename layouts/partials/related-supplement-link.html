{{/* Resolve a free-text synergy/conflict label to an actual supplement page. Renders when a page exists, plain when it doesn't — so broken descriptive labels like "Curcumin (enhanced anti-inflammatory)" stop 404ing. Tries (in order): 1. The whole label urlized 2. The substring before "(" urlized 3. The substring inside parentheses urlized */}} {{ $name := . }} {{ $candidates := slice ($name | urlize) }} {{ if strings.Contains $name "(" }} {{ $parts := split $name "(" }} {{ $before := index $parts 0 | strings.TrimSpace }} {{ if $before }}{{ $candidates = $candidates | append ($before | urlize) }}{{ end }} {{ $inside := index $parts 1 | replaceRE "\\).*$" "" | strings.TrimSpace }} {{ if $inside }}{{ $candidates = $candidates | append ($inside | urlize) }}{{ end }} {{ end }} {{ $found := "" }} {{ range $candidates }} {{ if not $found }} {{ with site.GetPage (printf "/supplements/%s" .) }}{{ $found = .RelPermalink }}{{ end }} {{ if not $found }}{{ with site.GetPage (printf "/peptides/%s" .) }}{{ $found = .RelPermalink }}{{ end }}{{ end }} {{ end }} {{ end }} {{ if $found -}} {{ $name }} {{- else -}} {{ $name }} {{- end -}}
