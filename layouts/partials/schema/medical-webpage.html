{{/* MedicalWebPage schema.org JSON-LD for /for// pages. Pulls symptom record from data/symptoms.json (matched by slug) and lists referenced supplements under `mentions` as MedicalSubstance. */}} {{- $slug := .File.BaseFileName -}} {{- if eq $slug "index" -}} {{- $slug = .Params.symptom_slug -}} {{- if not $slug -}}{{- $slug = path.Base (path.Dir .File.Path) -}}{{- end -}} {{- end -}} {{- $symptom := dict -}} {{- range $.Site.Data.symptoms.symptoms -}} {{- if eq .slug $slug -}}{{- $symptom = . -}}{{- end -}} {{- end -}} {{- $name := .Title -}} {{- $desc := "" -}} {{- with .Params.tagline -}}{{- $desc = . -}}{{- else -}}{{- with $symptom.description -}}{{- $desc = . -}}{{- else -}}{{- $desc = (.Summary | plainify | truncate 200) -}}{{- end -}}{{- end -}} {{- /* Build list of supplement records referenced by this symptom */ -}} {{- $supplements := slice -}} {{- range $symptom.supplements -}} {{- $id := . -}} {{- range $.Site.Data.supplements.supplements -}} {{- if eq .id $id -}}{{- $supplements = $supplements | append . -}}{{- end -}} {{- end -}} {{- end -}}
